ربط وحفظ البيانات تلقائياً في Google Sheet

يُقرأ التقرير مباشرة من رابط pubhtml/csv المرفق، ولإرسال التسجيلات الجديدة للشيت أدخل رابط Web App الخاص بك:

كود Apps Script للحفظ المباشر تلقائياً:

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(), data.school_name, data.school_stage,
    data.principal_name, data.principal_id, data.principal_phone, data.principal_qualification,
    data.vice_name, data.vice_id, data.vice_phone, data.vice_qualification,
    data.dev_name, data.dev_id, data.dev_phone, data.dev_position, data.dev_subject
  ]);
  return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON);
}
